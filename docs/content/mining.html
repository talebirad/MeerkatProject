How to Add a New Community Mining Algorithm to the Menu in MeerkatProject

This tutorial provides a step-by-step guide for adding a new community mining algorithm to the Community Mining menu in MeerkatProject. Follow these steps to add, configure, and integrate new algorithms that interact with the canvas and display clustering results.

1. Add the New Algorithm to english.xml

To make the new algorithm appear in the Community Mining menu, update MeerkatProject/MeerkatUI/resources/languages/english.xml.

Example XML Entry:

<MenuItem class="MenuItemNewAlgorithm" icon="" mnemonicParsing="false" text="New Algorithm" check="false" disabled="2" idmappingparameter="newalgorithm" SeparatorCount="0" />

This ensures the menu item is available in the UI.

2. Create a New Menu Item Class

Navigate to MeerkatProject/MeerkatUI/src/analysismenubar/ and create a new Java file named MenuItemNewAlgorithm.java.

Example Code:

package analysismenubar;

import analysisscreen.AnalysisController;
import communitymining.dialogwindow.NewAlgorithmDialog;
import javafx.scene.control.MenuItem;
import java.util.ArrayList;

public class MenuItemNewAlgorithm extends MenuItemGeneric implements IMenuItem {

    public MenuItemNewAlgorithm(String pstrMenuItemDisplay, String pstrMenuItemClass, String pstrMenuItemIconPath) {
        super(pstrMenuItemDisplay, pstrMenuItemClass, pstrMenuItemIconPath);
    }

    @Override
    public void Click(AnalysisController pController, MenuItem pMenuItem, Object pobjParameter) {
        // Open the new algorithm configuration dialog
        NewAlgorithmDialog.Display(pController, pobjParameter.toString(), new ArrayList<>());
    }
}

3. Implement the Algorithm's Configuration Dialog

Navigate to MeerkatProject/MeerkatUI/src/communitymining/dialogwindow/ and create NewAlgorithmDialog.java.

Example Code:

package communitymining.dialogwindow;

import analysisscreen.AnalysisController;
import config.LangConfig;
import config.StatusMsgsConfig;
import globalstate.MeerkatUI;
import java.util.List;
import javafx.scene.Scene;
import javafx.scene.control.Button;
import javafx.scene.control.ComboBox;
import javafx.scene.control.Label;
import javafx.scene.layout.GridPane;
import javafx.scene.layout.VBox;
import javafx.stage.Modality;
import javafx.stage.Stage;

public class NewAlgorithmDialog {

    public static void Display(AnalysisController pController, String pstrMappingID, List<Integer> VertexIDs) {
        try {
            Stage stgNewAlgorithm = new Stage();
            stgNewAlgorithm.initModality(Modality.APPLICATION_MODAL);

            // UI Components
            GridPane gridUserControls = new GridPane();
            Label lblParameter = new Label("Algorithm Parameter");
            ComboBox<String> cmbParameter = new ComboBox<>();
            cmbParameter.getItems().addAll("Option1", "Option2");
            cmbParameter.setValue("Option1");
            
            Button btnOK = new Button(LangConfig.GENERAL_OK);
            btnOK.setOnAction(e -> {
                // Execute Algorithm and Update Canvas
                String[] arrstrParameters = {"Parameter:" + cmbParameter.getValue()};
                stgNewAlgorithm.close();
                pController.updateStatusBar(true, StatusMsgsConfig.MINING_RESULTSCOMPUTING);
                
                MeerkatUI UIInstance = MeerkatUI.getUIInstance();
                int intProjectID = UIInstance.getActiveProjectTab().getProjectID();
                int intGraphID = UIInstance.getActiveProjectTab().getActiveGraphTab().getGraphID();
                int intTimeFrameIndex = UIInstance.getActiveProjectTab().getActiveGraphTab().getTimeFrameIndex();
                CommunityMiningAlgorithmDialog cmad = new CommunityMiningAlgorithmDialog();
                cmad.runAlgorithm(intProjectID, intGraphID, intTimeFrameIndex, pstrMappingID, arrstrParameters, pController);
            });

            VBox vboxPanel = new VBox(gridUserControls, btnOK);
            Scene scnNewAlgorithm = new Scene(vboxPanel);
            stgNewAlgorithm.setScene(scnNewAlgorithm);
            stgNewAlgorithm.show();
        } catch (Exception ex) {
            ex.printStackTrace();
        }
    }
}
